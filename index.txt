<script>
/* =========================================================================
   CONFIGURATION
   -------------------------------------------------------------------------
   1) SHOP_INFO_URL  -> raw GitHub URL of a .txt file with these lines:

        name=Your Shop Name
        logo=https://raw.githubusercontent.com/USER/REPO/main/logo.png
        google=https://g.page/r/XXXXXXXXXXXX/review
        instagram=https://instagram.com/yourshop
        whatsapp=https://wa.me/91XXXXXXXXXX
        facebook=https://facebook.com/yourshop

      - One "key=value" per line. Order doesn't matter.
      - Leave a value blank (or delete the line) to hide that button.
      - Must be the "raw" GitHub URL, e.g.:
        https://raw.githubusercontent.com/USER/REPO/main/shop-info.txt

   2) REVIEW_TEMPLATES_URL -> raw GitHub URL of a .txt file with one
      review template per line (blank lines are ignored). Put 1000+
      lines in there; 5 are picked at random each time.

        Loved the service here, staff were super friendly!
        Great quality and quick service, highly recommend.
        ... (1000+ lines)
   ========================================================================= */
  const CONFIG = {
    SHOP_INFO_URL: "https://raw.githubusercontent.com/USERNAME/REPO/main/shop-info.txt",
    REVIEW_TEMPLATES_URL: "https://raw.githubusercontent.com/USERNAME/REPO/main/review-templates.txt"
  };
/* ========================================================================= */

  let shopData = {};
  let allTemplates = [];
  let currentRating = null;

  const $ = (id) => document.getElementById(id);

  function showScreen(id){
    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    const target = $(id);
    target.classList.add('active');
    // restart stagger entrance animation each time the screen is shown
    target.querySelectorAll('.stagger').forEach(el => {
      el.style.animation = 'none';
      void el.offsetWidth;
      el.style.animation = '';
    });
  }

  function toast(msg){
    const t = $('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(() => t.classList.remove('show'), 2200);
  }

  /* ---------------- parse "key=value" txt into an object ---------------- */
  function parseShopInfo(text){
    const data = {};
    text.split('\n').forEach(line => {
      line = line.trim();
      if(!line || line.startsWith('#')) return;
      const idx = line.indexOf('=');
      if(idx === -1) return;
      const key = line.slice(0, idx).trim().toLowerCase();
      const value = line.slice(idx + 1).trim();
      if(value) data[key] = value;
    });
    return data;
  }

  /* ---------------- load shop info & build home screen ---------------- */
  async function loadShopInfo(){
    try{
      const res = await fetch(CONFIG.SHOP_INFO_URL, { cache: "no-store" });
      if(!res.ok) throw new Error("Could not load shop info (" + res.status + ")");
      shopData = parseShopInfo(await res.text());
      renderHome();
    }catch(err){
      $('shop-name').textContent = "Welcome";
      $('logo-wrap').innerHTML = '<span class="fallback">?</span>';
      $('home-error').style.display = 'block';
      $('home-error').textContent = "Couldn't load shop details. Check the GitHub link and try again.";
      console.error(err);
    }
  }

  function renderHome(){
    // name
    $('shop-name').textContent = shopData.name || "Our Shop";
    $('shop-eyebrow').style.visibility = 'visible';

    // logo
    const logoWrap = $('logo-wrap');
    if(shopData.logo){
      const img = new Image();
      img.onload = () => { logoWrap.innerHTML = ''; logoWrap.appendChild(img); };
      img.onerror = () => { logoWrap.innerHTML = '<span class="fallback">' + initials(shopData.name) + '</span>'; };
      img.src = shopData.logo;
      img.alt = shopData.name || "Shop logo";
    } else {
      logoWrap.innerHTML = '<span class="fallback">' + initials(shopData.name) + '</span>';
    }

    // buttons
    const wrap = $('home-buttons');
    wrap.innerHTML = '';

    if(shopData.google){
      wrap.appendChild(makeButton({
        primary: true,
        iconHtml: '✦',
        iconClass: '',
        title: 'Leave us a Google Review',
        sub: 'Takes less than a minute',
        onClick: () => showScreen('screen-rating')
      }));
    }
    if(shopData.whatsapp){
      wrap.appendChild(makeLinkButton({
        href: shopData.whatsapp, iconHtml: '💬', iconClass: 'whatsapp',
        title: 'Message us on WhatsApp'
      }));
    }
    if(shopData.instagram){
      wrap.appendChild(makeLinkButton({
        href: shopData.instagram, iconHtml: '📷', iconClass: 'instagram',
        title: 'Follow us on Instagram'
      }));
    }
    if(shopData.facebook){
      wrap.appendChild(makeLinkButton({
        href: shopData.facebook, iconHtml: '👍', iconClass: 'facebook',
        title: 'Visit our Facebook Page'
      }));
    }

    if(!wrap.children.length){
      wrap.innerHTML = '<p class="center-msg">No links configured yet.</p>';
    }
  }

  function initials(name){
    if(!name) return "🏬";
    return name.trim().split(/\s+/).slice(0,2).map(w => w[0].toUpperCase()).join('');
  }

  function makeButton({primary, iconHtml, iconClass, title, sub, onClick}){
    const b = document.createElement('button');
    b.className = 'btn' + (primary ? ' primary' : '');
    b.innerHTML = `<span class="icon ${iconClass}">${iconHtml}</span>
                   <span><span class="title">${title}</span>${sub ? `<span class="sub">${sub}</span>` : ''}</span>`;
    b.onclick = onClick;
    return b;
  }

  function makeLinkButton({href, iconHtml, iconClass, title}){
    const a = document.createElement('a');
    a.className = 'btn';
    a.href = href;
    a.target = '_blank';
    a.rel = 'noopener noreferrer';
    a.innerHTML = `<span class="icon ${iconClass}">${iconHtml}</span><span class="title">${title}</span>`;
    return a;
  }

  function goHome(){ showScreen('screen-home'); }
  function goToRating(){ showScreen('screen-rating'); }

  /* ---------------- rating chosen -> load templates ---------------- */
  async function chooseRating(rating){
    currentRating = rating;
    showScreen('screen-templates');
    $('refresh-btn').style.display = 'none';
    $('templates-list').innerHTML = '<div class="spinner"></div>';

    try{
      if(allTemplates.length === 0){
        const res = await fetch(CONFIG.REVIEW_TEMPLATES_URL, { cache: "no-store" });
        if(!res.ok) throw new Error("Could not load templates (" + res.status + ")");
        const text = await res.text();
        allTemplates = text.split('\n').map(l => l.trim()).filter(Boolean);
      }
      if(allTemplates.length === 0){
        $('templates-list').innerHTML = '<p class="error-msg">No review templates found yet.</p>';
        return;
      }
      renderTemplates();
      $('refresh-btn').style.display = 'block';
    }catch(err){
      $('templates-list').innerHTML = '<p class="error-msg">Couldn\'t load review templates. Check the GitHub link and try again.</p>';
      console.error(err);
    }
  }

  function pickRandom(arr, n){
    const copy = arr.slice();
    const picked = [];
    n = Math.min(n, copy.length);
    for(let i = 0; i < n; i++){
      const idx = Math.floor(Math.random() * copy.length);
      picked.push(copy[idx]);
      copy.splice(idx, 1);
    }
    return picked;
  }

  function renderTemplates(){
    const chosen = pickRandom(allTemplates, 5);
    const list = $('templates-list');
    list.innerHTML = '';
    chosen.forEach(text => {
      const card = document.createElement('div');
      card.className = 'template-card';
      card.innerHTML = `${escapeHtml(text)}<span class="tap-hint">Tap to copy &amp; post on Google &rarr;</span>`;
      card.onclick = () => useTemplate(text);
      list.appendChild(card);
    });
  }

  function reshuffleTemplates(){ renderTemplates(); }

  function escapeHtml(str){
    const d = document.createElement('div');
    d.textContent = str;
    return d.innerHTML;
  }

  async function useTemplate(text){
    try{
      await navigator.clipboard.writeText(text);
      toast('Review copied! Paste it on the next screen.');
    }catch(err){
      toast('Opening Google review — copy the text above first.');
    }
    setTimeout(() => {
      if(shopData.google) window.open(shopData.google, '_blank', 'noopener,noreferrer');
    }, 500);
  }

  /* ---------------- init ---------------- */
  loadShopInfo();
</script>
